import streamlit as st import os import chromadb from langchain_community.document_loaders import PyPDFLoader from langchain_text_splitters import RecursiveCharacterTextSplitter from langchain_community.embeddings import FastEmbedEmbeddings from langchain_community.vectorstores import Chroma from langchain_community.llms import Ollama from langchain.chains import RetrievalQA st.set_page_config(page_title="Shakthi AI • Sovereign Edge", layout="wide", page_icon="⚡") st.markdown("<h2 style='color:#0284c7;'>⚡ SHAKTHI AI — Sovereign Edge Appliance</h2>", unsafe_allow_html=True) st.markdown("<p style='color:#64748b;'>100% Offline Vernacular Intelligence • Karnataka DSERT Curriculum & Health Knowledge Base</p>", unsafe_allow_html=True) with st.sidebar: st.markdown("### 🖥️ Edge Appliance Status") st.success("● 100% AIR-GAPPED OFFLINE") st.markdown("**Compute Engine:** Quad-Core CPU (8GB RAM)") st.markdown("**Vector Store:** ChromaDB Local Persistent") st.markdown("**Inference Model:** LLaMA 3.2 3B (4-bit Quantized)") st.markdown("**Knowledge Anchor:** DSERT SSLC + NHM WIFS") st.divider() st.markdown("Powered by **NAMMA WEB**") tab1, tab2, tab3 = st.tabs([ "💬 Vernacular Student Assistant (ಕನ್ನಡ)", "📊 ICDS Frontline Telemetry", "📁 Document Knowledge Manager" ]) DB_DIR = "./chroma_db_shakthi" embeddings = FastEmbedEmbeddings(model_name="BAAI/bge-small-en-v1.5") if not os.path.exists(DB_DIR): with st.spinner("Pre-indexing Shakthi Master Dataset..."): loader = PyPDFLoader("Shakthi_AI_Ollama_Kannada_Master_Dataset.pdf") docs = loader.load() splitter = RecursiveCharacterTextSplitter(chunk_size=350, chunk_overlap=40) chunks = splitter.split_documents(docs) vectordb = Chroma.from_documents(chunks, embeddings, persist_directory=DB_DIR) else: vectordb = Chroma(persist_directory=DB_DIR, embedding_function=embeddings) llm = Ollama(model="llama3.2:latest", temperature=0.1) retriever = vectordb.as_retriever(search_kwargs={"k": 2}) qa_chain = RetrievalQA.from_chain_type( llm=llm, chain_type="stuff", retriever=retriever, return_source_documents=True ) with tab1: user_query = st.chat_input("ಕನ್ನಡದಲ್ಲಿ ಪ್ರಶ್ನೆ ಕೇಳಿ (Ask regarding Anemia, Diet, SSLC Science, or Safety)...") if user_query: with st.chat_message("user"): st.write(user_query) with st.chat_message("assistant"): with st.spinner("ಯೋಚಿಸುತ್ತಿದೆ..."): response = qa_chain.invoke({"query": user_query}) st.write(response["result"]) with tab2: st.subheader("📈 Anonymized Taluk Health & Compliance Pulse") colA, colB, colC = st.columns(3) colA.metric("Pilot School Adherence", "84.2%", "+18% vs Baseline") colB.metric("Top Student Query", "IFA Stool Color / Nausea", "Resolved via AI") colC.metric("Dropout Risk Index", "Low (12%)", "-35% Intervention Alert") st.info("💡 Store-and-Forward Telemetry: Syncs anonymized cohort trends to Taluk Health Officers when hotspot is available.") with tab3: st.subheader("📥 Add Verified Local Knowledge Manuals") uploaded_file = st.file_uploader("Upload DSERT or Clinical PDF", type=["pdf"]) if uploaded_file: st.success(f"File '{uploaded_file.name}' staged for local embedding ingestion.")
